-- breaks ties on MIN(placement), which is 1 for anyone who won a single game,

SELECT COUNT(*) INTO @s3_tied
FROM v_public_standings a
JOIN v_public_standings b
  ON b.tournament_id = a.tournament_id
 AND b.standing      = a.standing
 AND b.competitor   <> a.competitor
WHERE a.tournament_id = 5

SELECT COUNT(*) INTO @s13_inv
FROM (
    SELECT c.competitor_id
         , SUM(mp.points) AS pts
         , SUM(ps.score)  AS strokes
    FROM MatchParticipant mp
    JOIN `Match` m           ON m.match_id      = mp.match_id
    JOIN Competitor c        ON c.competitor_id = mp.competitor_id
    JOIN PlayerMatchStats ps ON ps.match_id     = mp.match_id
                            AND ps.player_id    = c.player_id
    WHERE m.tournament_id = 7
    GROUP BY c.competitor_id
    HAVING COUNT(*) = 4
) a
JOIN (
    SELECT c.competitor_id
         , SUM(mp.points) AS pts
         , SUM(ps.score)  AS strokes
    FROM MatchParticipant mp
    JOIN `Match` m           ON m.match_id      = mp.match_id
    JOIN Competitor c        ON c.competitor_id = mp.competitor_id
    JOIN PlayerMatchStats ps ON ps.match_id     = mp.match_id
                            AND ps.player_id    = c.player_id
    WHERE m.tournament_id = 7
    GROUP BY c.competitor_id
    HAVING COUNT(*) = 4
) b ON b.competitor_id > a.competitor_id
WHERE (a.pts > b.pts AND a.strokes > b.strokes)
   OR (a.pts < b.pts AND a.strokes < b.strokes)

| S11  | TENNIS     | the loaded tennis scores were kept short enough to fit, which is a property of the matches we chose, not of the schema                                                                                                             |

| S16  | GOLF       | every round loaded above was given artificially distinct scores so that the file would load in both modes                                                                                                                          |

| S21  | CROSS      | the entity is called Game and holds Basketball; every view labels the column "game" to an audience reading a basketball schedule. Pure naming, zero structural cost, but it is a lie the ERD tells                                 |
