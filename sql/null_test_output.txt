-- WHAT IS BEING TESTED
--------------

--------------
-- An inner join on a NULLable column silently discards the rows whose join
--------------

--------------
-- value is NULL: NULL = NULL is UNKNOWN, never TRUE. The row still exists in
--------------

--------------
-- the base table, the user is simply never shown it and gets no warning. The
--------------

--------------
-- same three-valued logic bites twice more in our code:
--------------

--------------
--   * a WHERE predicate such as  status <> 'paid'  drops NULL status;
--------------

--------------
--   * NULL arithmetic (a - b, a / b) poisons a whole computed column.
--------------

--------------
-- All three families are covered below.
--------------

--------------
--
--------------

--------------
-- METHOD -- the same rule as 08_acid_tests.sql
--------------

--------------
-- Every verdict is computed by a real COUNT(*) or SUM() against real rows,
--------------

--------------

--------------

-- inline null-safe rewrite of it side by side and compares the two row counts.
--------------

-- The difference IS the number of dangling tuples. Nothing is dropped, altered
--------------

SELECT COUNT(*), COALESCE(SUM(total_expense), 0)
  INTO @n1_rows_view, @n1_exp_view

-- what a null-safe rewrite shows
--------------

--------------
SELECT COUNT(*), COALESCE(SUM(x.total_expense), 0)
  INTO @n1_rows_fix, @n1_exp_fix
FROM (
    SELECT t.tournament_id
         , SUM(CASE WHEN tr.type = 'expense' THEN tr.amount ELSE 0 END) AS total_expense
    FROM Transactions tr
    LEFT JOIN Tournament t ON t.tournament_id = tr.tournament_id
    JOIN Membership mb     ON mb.org_id       = tr.org_id
    JOIN Users u           ON u.user_id       = mb.user_id
    WHERE u.db_username = SUBSTRING_INDEX(SESSION_USER(), '@', 1)
      AND mb.role = 'admin'
      AND mb.left_date IS NULL
    GROUP BY t.tournament_id
) x

ROLLBACK
--------------

--------------

('N1', 'view', 'v_org_financials', IF(@n1_exp_fix > @n1_exp_view, 'GAP', 'OK'),
